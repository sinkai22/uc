package practical8;
import dk.pervasive.jcaf.ContextService;
import dk.pervasive.jcaf.impl.ContextServiceImpl;
import dk.pervasive.jcaf.item.Location;
import dk.pervasive.jcaf.relationship.Located;
public class ContextAwareSystem {
    public static void main(String[] args) {
        try {
            // Create a person entity
            Person person = new Person("person1", "Harshita");
            System.out.println("Person created.");
            System.out.println("Name: " + person.getName());
            System.out.println("ID: " + person.getId());
            // Get the JCAF Context Service
            ContextService contextService =
                    new ContextServiceImpl("practical8");
            // Register the person with JCAF
            contextService.addEntity(person);
            System.out.println();
            System.out.println("Person registered with JCAF.");
            System.out.println();
            // Create a location
            Location location =
                    new Location("Computer_Lab");
            // Create a Located relationship
            Located located =
                    new Located("Practical_8");
            // Add location context to the person
            contextService.addContextItem(
                    person.getId(),
                    located,
                    location
            );
            System.out.println();
            System.out.println("Location context added.");
            System.out.println("Location: Computer_Lab");
        } catch (Exception e) {
            e.printStackTrace();
        }
}
    }
